        async function checkStudentStatus() {
            const studentId = document.getElementById('studentIdInput').value;
            const password = document.getElementById('adminPasswordInput').value;
            const semester = document.getElementById('semesterSelect').value;
            const resultBox = document.getElementById('profileResult');
            
            if (!studentId) { alert('Please enter Student ID'); return; }

            resultBox.style.display = 'block';
            resultBox.className = 'result loading';
            resultBox.innerHTML = '<p>⏳ Checking...</p>';

            try {
                const response = await fetch(CHECK_STATUS_URL, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ student_id: studentId, semester: semester })
                });
                const data = await response.json();
                
                if (response.ok) {
                    currentStudentId = studentId;
                    resultBox.className = 'result success';
                    resultBox.innerHTML = `<h3>✅ ${data.message}</h3><p>Name: ${data.student_name} | School: ${data.school_id} | Semester: ${semester}</p>`;
                    
                    // Show standard sections
                    document.getElementById('schoolProfileSection').style.display = 'block';
                    document.getElementById('srjSection').style.display = 'block';
                    document.getElementById('reportSection').style.display = 'block';
                    
                    // ADMIN SECURITY CHECK - esayat88 ONLY
                    if (currentStudentId === ADMIN_ID) {
                        if (password === ADMIN_PASSWORD) {
                            document.getElementById('adminSection').style.display = 'block';
                        } else {
                            document.getElementById('adminSection').style.display = 'none';
                            alert('⚠️ Incorrect Admin Password. Administrator dashboard access denied.');
                        }
                    } else {
                        document.getElementById('adminSection').style.display = 'none';
                    }
                } else {
                    resultBox.className = 'result error';
                    resultBox.innerHTML = `<h3>❌ ${data.message}</h3>`;
                    document.getElementById('adminSection').style.display = 'none';
                }
            } catch (error) {
                resultBox.className = 'result error';
                resultBox.innerHTML = `<h3>🔌 Error</h3><p>${error.message}</p>`;
                document.getElementById('adminSection').style.display = 'none';
            }
        }

        function captureGPS() {
            const statusDiv = document.getElementById('regResult');
            statusDiv.style.display = 'block';
            statusDiv.className = 'result loading';
            statusDiv.innerHTML = '<p>📡 Acquiring satellite signal...</p>';

            if (!navigator.geolocation) {
                statusDiv.className = 'result error';
                statusDiv.innerHTML = 'Geolocation not supported by your browser.';
                return;
            }

            navigator.geolocation.getCurrentPosition(
                (position) => {
                    document.getElementById('regLat').value = position.coords.latitude.toFixed(6);
                    document.getElementById('regLong').value = position.coords.longitude.toFixed(6);
                    statusDiv.className = 'result success';
                    statusDiv.innerHTML = `✅ GPS Captured! Lat: ${position.coords.latitude.toFixed(4)}, Long: ${position.coords.longitude.toFixed(4)}`;
                },
                (error) => {
                    statusDiv.className = 'result error';
                    statusDiv.innerHTML = `❌ GPS Error: ${error.message}. Please enable location services.`;
                },
                { enableHighAccuracy: true, timeout: 10000 }
            );
        }

        async function captureAttendancePhoto() {
            const resultDiv = document.getElementById('photoResult');
            const video = document.getElementById('camera');
            const canvas = document.getElementById('canvas');
            const photoBtn = document.getElementById('photoBtn');
            
            const now = new Date();
            const hour = now.getHours();
            const minute = now.getMinutes();
            const currentMinutes = hour * 60 + minute;
            const startMinutes = 10 * 60 + 30;  // 10:30 AM
            const endMinutes = 12 * 60;          // 12:00 PM
            
            if (currentMinutes < startMinutes || currentMinutes > endMinutes) {
                resultDiv.style.display = 'block';
                resultDiv.className = 'result error';
                resultDiv.innerHTML = `<strong>⚠️ Wrong Time!</strong> Photo capture only allowed between 10:30 AM and 12:00 PM.<br>Current time: ${hour}:${minute.toString().padStart(2, '0')}`;
                return;
            }

            photoBtn.disabled = true;
            photoBtn.textContent = '📷 Initializing camera...';
            resultDiv.style.display = 'block';
            resultDiv.className = 'result loading';
            resultDiv.innerHTML = '<p>📷 Initializing camera...</p>';

            try {
                const stream = await Promise.race([
                    navigator.mediaDevices.getUserMedia({ 
                        video: { facingMode: 'user', width: { ideal: 640 }, height: { ideal: 480 } } 
                    }),
                    new Promise((_, reject) => setTimeout(() => reject(new Error('Camera access timeout')), 10000))
                ]);
                
                video.style.display = 'block';
                video.srcObject = stream;
                
                await new Promise((resolve) => {
                    video.onloadedmetadata = () => { video.play(); resolve(); };
                });
                
                photoBtn.textContent = 'Taking photo in 3 seconds...';
                resultDiv.innerHTML = '<p>📸 Position yourself and smile! Photo will be taken in 3 seconds...</p>';

                for (let i = 3; i > 0; i--) {
                    resultDiv.innerHTML += `<p>${i}...</p>`;
                    await new Promise(resolve => setTimeout(resolve, 1000));
                }

                const context = canvas.getContext('2d');
                canvas.width = video.videoWidth || 640;
                canvas.height = video.videoHeight || 480;
                context.drawImage(video, 0, 0, canvas.width, canvas.height);
                const photoData = canvas.toDataURL('image/jpeg', 0.8);
                
                stream.getTracks().forEach(track => track.stop());
                video.style.display = 'none';
                
                resultDiv.className = 'result loading';
                resultDiv.innerHTML = '<p> Uploading photo... Please wait...</p>';
                
                const uploadPromise = fetch(CAPTURE_ATTENDANCE_PHOTO_URL, {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({
                        student_id: currentStudentId,
                        photo_data: photoData,
                        week: parseInt(document.getElementById('weekNumber').value),
                        day: parseInt(document.getElementById('dayNumber').value),
                        timestamp: new Date().toISOString()
                    })
                });
                
                const timeoutPromise = new Promise((_, reject) => 
                    setTimeout(() => reject(new Error('Upload timeout - please try again')), 30000)
                );
                
                const response = await Promise.race([uploadPromise, timeoutPromise]);
                const result = await response.json();
                
                if (response.ok) {
                    resultDiv.className = 'result success';
                    resultDiv.innerHTML = `<strong>✅ Photo Captured Successfully!</strong><br>
                        <p>Your attendance at ${hour}:${minute.toString().padStart(2, '0')} has been verified.</p>
                        <p style="margin-top:10px; font-size:12px;">Week ${document.getElementById('weekNumber').value}, Day ${document.getElementById('dayNumber').value}</p>`;
                } else {
                    throw new Error(result.error || 'Upload failed');
                }
                
            } catch (error) {
                console.error('Photo capture error:', error);
                resultDiv.className = 'result error';
                
                let errorMessage = '❌ <strong>Camera Error:</strong> ';
                if (error.name === 'NotAllowedError' || error.message.includes('permission')) {
                    errorMessage += 'Camera access denied. Please allow camera permissions.';
                } else if (error.name === 'NotFoundError' || error.message.includes('device')) {
                    errorMessage += 'No camera found.';
                } else if (error.message.includes('timeout')) {
                    errorMessage += error.message;
                } else {
                    errorMessage += error.message + '. Please try again.';
                }
                
                resultDiv.innerHTML = errorMessage;
            } finally {
                photoBtn.textContent = '📷 Capture Attendance Photo (10:30 AM - 12:00 PM)';
                photoBtn.disabled = false;
                if (video.srcObject) {
                    video.srcObject.getTracks().forEach(track => track.stop());
                }
                video.style.display = 'none';
            }
        }
